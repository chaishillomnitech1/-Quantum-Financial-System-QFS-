## 📖 Usage

### Quick Start Example

```python
from qfs import (
    QuantumBlockchain,
    RoseGoldEncryption,
    WealthDistributor,
    NESARAGESARACompliance,
    GalacticFederationFramework
)

# Initialize quantum blockchain
blockchain = QuantumBlockchain(difficulty=4)

# Add and mine transactions
blockchain.add_transaction({
    "from": "treasury",
    "to": "citizen_001",
    "amount": 1000,
    "type": "universal_basic_income"
})
blockchain.mine_pending_transactions("miner_001")

# Use Rose Gold encryption
encryption = RoseGoldEncryption()
encrypted_data = encryption.encrypt("Sensitive financial data")
decrypted_data, verified = encryption.decrypt(encrypted_data)

# Distribute wealth
from qfs.wealth_distribution.distribution_algorithm import Participant, DistributionStrategy

participants = [
    Participant(address="citizen_001", stake=100, need_score=0.8),
    Participant(address="citizen_002", stake=200, need_score=0.5),
]

distributor = WealthDistributor()
allocation = distributor.distribute(10000, participants, DistributionStrategy.GALACTIC_ABUNDANCE)

# Check NESARA/GESARA compliance
compliance = NESARAGESARACompliance()
report = compliance.check_compliance("entity_001", {
    "from": "treasury",
    "to": "citizen_001",
    "amount": 5000,
    "type": "debt_forgiveness"
})

# Setup galactic framework
from qfs.galactic_framework.federation_framework import DimensionType

framework = GalacticFederationFramework()
framework.deposit_to_galactic_treasury(1000000)
framework.anchor_abundance_portal("portal_alpha", DimensionType.QUANTUM, 50000)
```
